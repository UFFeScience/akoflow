Cloud screens use a separate local fixture without provider credentials; its provisioning attempt fails before any cloud resources are created. Some list captures show the empty state from before the fixtures were loaded. The instance name shown in the status bar is a demonstration label.

export function ScreenGallery({group}) {
  const baseUrl = useBaseUrl('/');
  return (
    <div>
      {screens.filter((screen) => screen.group === group).map((screen) => (
        <section key={screen.path}>
          <h3>{screen.title}</h3>
          <p>{screen.detail} Open <code>{screen.route}</code> in Desktop.</p>
          <img src={`${baseUrl}img/interface/${screen.path}`} alt={`AkôFlow Desktop ${screen.title} screen: ${screen.detail}`} loading="lazy" />
        </section>
      ))}
    </div>
  );
}
